# FakePlayerPlugin — Skin Folder
#
# Place PNG skin files here to use them for bots.
# Requires: skin.mode = custom  in config.yml
#
# Naming rules:
#   <botname>.png  — assigned exclusively to the bot with that name
#   anything.png   — added to the random skin pool
#
# Skin files must be standard 64x64 or 64x32 Minecraft skin PNGs.
# Run /fpp reload after adding or removing skin files.
